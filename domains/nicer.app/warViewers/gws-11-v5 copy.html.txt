  ring.forEach(([lon,lat],idx)=>{
    const x=(lon+180)/360*W;
    const y=(90-lat)/180*H;
    // Prevent anti-meridian horizontal wrap streaks
    if(idx===0 || (prevLon!==null && Math.abs(lon-prevLon)>180)){
      ctx.moveTo(x,y);
    }else{
      ctx.lineTo(x,y);
    }
    prevLon=lon;
  });
  ctx.closePath();
}

function build3DCountryBorders(geoFeatures){
  while(borderGroup.children.length) borderGroup.remove(borderGroup.children[0]);
  const positions=[];
  const addRing=(ring)=>{
    for(let i=0;i<ring.length-1;i++){
      const [lon1,lat1]=ring[i];
      const [lon2,lat2]=ring[i+1];
      if(Math.abs(lon1-lon2)>180) continue;
      const v1=ll2v(lat1,lon1,0.08);
      const v2=ll2v(lat2,lon2,0.08);
      positions.push(v1.x,v1.y,v1.z, v2.x,v2.y,v2.z);
    }
  };
  geoFeatures.forEach(f=>{
    const g=f.geometry;
    if(!g) return;
    if(g.type==="Polygon") g.coordinates.forEach(addRing);
    else if(g.type==="MultiPolygon") g.coordinates.forEach(poly=>poly.forEach(addRing));
  });
  const geo=new THREE.BufferGeometry();
  geo.setAttribute("position",new THREE.Float32BufferAttribute(positions,3));
  const mat=new THREE.LineBasicMaterial({color:0x00ffcc,transparent:true,opacity:0.45});
  borderGroup.add(new THREE.LineSegments(geo,mat));
}

function isLand(lat,lon){
  if(!mapData) return true;
  const normLon=((lon+540)%360)-180;
  const x=Math.max(0,Math.min(W-1,Math.floor((normLon+180)/360*W)));
  const y=Math.max(0,Math.min(H-1,Math.floor((90-lat)/180*H)));
  return mapData[(y*W+x)*4+1] > 32;
}

function ll2v(lat,lon,alt=0){
  const phi=(90-lat)*Math.PI/180;
  const theta=(lon+180)*Math.PI/180;
  const r=ER+alt;
  return new THREE.Vector3(-r*Math.sin(phi)*Math.cos(theta), r*Math.cos(phi), r*Math.sin(phi)*Math.sin(theta));
}

function greatCircleDist(lat1,lon1,lat2,lon2){
  const p1=lat1*Math.PI/180, p2=lat2*Math.PI/180;
  const dp=(lat2-lat1)*Math.PI/180, dl=(lon2-lon1)*Math.PI/180;
  const a=Math.sin(dp/2)**2+Math.cos(p1)*Math.cos(p2)*Math.sin(dl/2)**2;
  return 2*Math.atan2(Math.sqrt(a),Math.sqrt(1-a))*180/Math.PI;
}

function placeTangent(mesh, lat, lon, alt=0.35){
  const pos = ll2v(lat, lon, alt);
  mesh.position.copy(pos);
  const normal = pos.clone().normalize();
  mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 0, 1), normal);
}

function launchMissile(fromObj,toObj,isTactical=false){
  if(fromObj.hp<=0 || toObj.hp<=0) return;
  const start=ll2v(fromObj.lat,fromObj.lon,0.5);
  const end=ll2v(toObj.lat,toObj.lon,0.5);
  const dist=start.distanceTo(end);
  const peak=ER + (isTactical ? dist*0.18 : Math.min(26, dist*0.35));
  const mid=start.clone().add(end).multiplyScalar(0.5).normalize().multiplyScalar(peak);

  const curve=new THREE.QuadraticBezierCurve3(start,mid,end);
  const pts=curve.getPoints(48);
  const colorHex=fromObj.side==="N"?0x4da6ff:0xff4d4d;

  const guideGeo=new THREE.BufferGeometry().setFromPoints(pts);
  const guideMat=new THREE.LineBasicMaterial({
    color:colorHex,
    transparent:true,
    opacity:0.28
  });
  const guideLine=new THREE.Line(guideGeo,guideMat);
  simGroup.add(guideLine);

  const trailGeo=new THREE.BufferGeometry().setFromPoints([start, start.clone()]);
  const trailMat=new THREE.LineBasicMaterial({
    color:fromObj.side==="N"?0x80d4ff:0xff8080,
    transparent:true,
    opacity:0.95
  });
  const trailLine=new THREE.Line(trailGeo,trailMat);
  simGroup.add(trailLine);

  const arrowMat=new THREE.MeshBasicMaterial({color:fromObj.side==="N"?0x99e6ff:0xff9999});
  const arrow=new THREE.Mesh(arrowGeo,arrowMat);
  arrow.position.copy(start);
  arrow.lookAt(curve.getPoint(0.05));
  simGroup.add(arrow);

  missiles.push({
    from:fromObj,to:toObj,side:fromObj.side,
    curve,pts,t:0,
    baseSpeed:(isTactical?0.0085:0.0048),
    dmg:isTactical?18:34,
    guideLine,trailLine,arrow
  });
}

function cleanupMissile(m){
  simGroup.remove(m.guideLine);
  simGroup.remove(m.trailLine);
  simGroup.remove(m.arrow);
  m.guideLine.geometry.dispose();
  m.trailLine.geometry.dispose();
}

function triggerStrike(attackerSide){
  if(state==="FINISHED") initGrid();
  state="ENGAGED";
  $("hs").textContent="ACTIVE COMBAT"; $("hs").style.color="#ff4d4d";
  $("bN").disabled=true; $("bB").disabled=true;

  const atkBases=bases.filter(b=>b.side===attackerSide && b.hp>0);
  const defTargets=[...bases,...units].filter(o=>o.side!==attackerSide && o.hp>0);

  logMsg("ALERT: "+(attackerSide==="N"?"NATO":"BRICS+")+" initiated multi-domain first strike!");
  atkBases.slice(0,18).forEach(b=>{
    const tgt=defTargets[Math.floor(Math.random()*defTargets.length)];
    if(tgt) launchMissile(b,tgt,false);
  });

  setTimeout(()=>{
    if(state!=="ENGAGED") return;
    const retSide=attackerSide==="N"?"B":"N";
    const retBases=bases.filter(b=>b.side===retSide && b.hp>0);
    const retUnits=units.filter(u=>u.side===retSide && u.hp>0);
    const oppTargets=[...bases,...units].filter(o=>o.side===attackerSide && o.hp>0);
    logMsg("RETALIATION: "+(retSide==="N"?"NATO":"BRICS+")+" launching counter-salvo from silos & fleets!");
    [...retBases.slice(0,14),...retUnits.slice(0,8)].forEach(src=>{
      const tgt=oppTargets[Math.floor(Math.random()*oppTargets.length)];
      if(tgt) launchMissile(src,tgt,src.kind!=="BASE");
    });
  },2600);
}

function updateHUD(){
  const calcAvg=(arr,side)=>{
    const sub=arr.filter(x=>x.side===side);
    return sub.length?Math.round(sub.reduce((a,b)=>a+b.hp,0)/sub.length):0;
  };
  const nBases=bases.filter(b=>b.side==="N"), bBases=bases.filter(b=>b.side==="B");
  const nSea=units.filter(u=>u.side==="N"&&u.dom==="SEA"), bSea=units.filter(u=>u.side==="B"&&u.dom==="SEA");
  const nLand=units.filter(u=>u.side==="N"&&u.dom==="LAND"), bLand=units.filter(u=>u.side==="B"&&u.dom==="LAND");

  const nB=calcAvg(nBases,"N"), nS=calcAvg(nSea,"N"), nL=calcAvg(nLand,"N");
  const bB=calcAvg(bBases,"B"), bS=calcAvg(bSea,"B"), bL=calcAvg(bLand,"B");

  $("hn").textContent=nB+"% | "+nS+"% | "+nL+"%";
  $("hb").textContent=bB+"% | "+bS+"% | "+bL+"%";
  $("hv").textContent=missiles.length+" / "+units.filter(u=>u.hp>0).length;
  $("hh").textContent=hitCount+" / "+intCount;
  $("hc").textContent=clashCount;

  if(state==="ENGAGED"){
    const nTotal=(nB+nS+nL)/3, bTotal=(bB+bS+bL)/3;
    if(nTotal<18 || bTotal<18){
      state="FINISHED";
      $("hs").textContent="SIMULATION COMPLETE"; $("hs").style.color="#00ffcc";
      $("vf").style.display="flex";
      const winner=nTotal>bTotal?"NATO / ALLIED COALITION":"BRICS+ COALITION";
      $("vl").textContent=winner;
      logMsg("ENDEX: Strategic dominance achieved by "+winner+".");
    }
  }
}

let lastUI=0;
function animate(now){
  requestAnimationFrame(animate);
  controls.update();

  const simSpd = Number($("simSpd").value);
  const machMult = Number($("mach").value) / 8;

  if(state==="ENGAGED"){
    const ewChance=Number($("ew").value)/100;
    const dipN=Number($("dipN").value);
    const dipB=Number($("dipB").value);

    dipTimer += 0.016 * simSpd;
    if(dipTimer >= 6.0){
      dipTimer = 0;
      const jointDip = (dipN * dipB) / 10000;
      if(jointDip > 0.12 && Math.random() < jointDip * 0.55){
        state="FINISHED";
        missiles.forEach(cleanupMissile);
        missiles=[];
        $("hs").textContent="CEASEFIRE SIGNED"; $("hs").style.color="#00ffcc";
        $("vf").style.display="flex";
        $("vl").textContent="NEGOTIATED ARMISTICE";
        logMsg("DIPLOMACY: Backchannel talks succeeded (NATO "+dipN+"% / BRICS "+dipB+"%). Immediate ceasefire!");
        updateHUD();
        renderRegistry();
      } else if (dipN >= 70 || dipB >= 70) {
        if(Math.random() < 0.35) {
          logMsg("DIPLOMACY: Envoys from "+(dipN>=dipB?"NATO":"BRICS+")+" proposing de-escalation corridor...");
        }
      }
    }

    for(let i=missiles.length-1;i>=0;i--){
      const m=missiles[i];
      m.t += m.baseSpeed * machMult * simSpd;
      const clampedT = Math.min(1, m.t);

      const pos = m.curve.getPoint(clampedT);
      m.arrow.position.copy(pos);
      if(clampedT < 0.99){
        const nextPos = m.curve.getPoint(Math.min(1, clampedT + 0.02));
        m.arrow.lookAt(nextPos);
      }

      const sliceCount = Math.max(2, Math.floor(clampedT * m.pts.length));
      const trailPts = m.pts.slice(0, sliceCount);
      trailPts.push(pos);
      m.trailLine.geometry.dispose();
      m.trailLine.geometry = new THREE.BufferGeometry().setFromPoints(trailPts);

      if(m.t>=1){
        cleanupMissile(m);
        missiles.splice(i,1);

        const defenders=units.filter(u=>u.hp>0 && u.side!==m.side && greatCircleDist(u.lat,u.lon,m.to.lat,m.to.lon)<14);
        const interceptProb=Math.min(0.85, ewChance + defenders.length*0.14);

        if(Math.random()<interceptProb){
          intCount++;
          spawnExplosion(m.to.lat,m.to.lon,0x00ffcc);
          if(Math.random()<0.25) logMsg("INTERCEPT: "+(defenders.length?defenders[0].name:"EW Grid")+" neutralized warhead near "+m.to.name+".");
        }else{
          hitCount++;
          m.to.hp=Math.max(0,m.to.hp-m.dmg);
          spawnExplosion(m.to.lat,m.to.lon,0xff4400);
          if(m.to.hp===0){
            m.to.mesh.visible=false;
            logMsg("DESTROYED: "+m.to.name+" neutralized!");
          }
        }
      }
    }

    units.forEach(u=>{
      if(u.hp<=0) return;
      const sideDip = u.side==="N" ? dipN : dipB;
      const aggressionFactor = Math.max(0.2, 1 - (sideDip / 130));
      u.cd -= 0.016 * simSpd * aggressionFactor;

      const dLat=u.tLat-u.lat, dLon=u.tLon-u.lon;
      const dist=Math.hypot(dLat,dLon);
      if(dist>0.8){
        const stepLat=u.lat+(dLat/dist)*u.speed*simSpd;
        const stepLon=u.lon+(dLon/dist)*u.speed*simSpd;
        const terrainOk=(u.dom==="LAND") ? isLand(stepLat,stepLon) : !isLand(stepLat,stepLon);
        if(terrainOk){
          u.lat=stepLat; u.lon=stepLon;
        }else{
          u.tLat+=(Math.random()-0.5)*8;
          u.tLon+=(Math.random()-0.5)*8;
        }
        placeTangent(u.mesh, u.lat, u.lon, 0.45);
      }else{
        const enemies=[...bases,...units].filter(e=>e.hp>0 && e.side!==u.side);
        if(enemies.length){
          const next=enemies[Math.floor(Math.random()*enemies.length)];
          u.tLat=next.lat; u.tLon=next.lon;
        }
      }

      if(u.cd<=0){
        u.cd=3.5+Math.random()*3;
        const nearbyEnemies=[...units,...bases].filter(e=>e.hp>0 && e.side!==u.side && greatCircleDist(u.lat,u.lon,e.lat,e.lon)<16);
        if(nearbyEnemies.length){
          const target=nearbyEnemies[0];
          const range=greatCircleDist(u.lat,u.lon,target.lat,target.lon);
          if(range<6.5){
            clashCount++;
            const dmg=12+Math.random()*14;
            target.hp=Math.max(0,target.hp-dmg);
            spawnExplosion(target.lat,target.lon,0xffff00);
            logMsg(u.dom+" CLASH: "+u.name+" engaged "+target.name+" (-"+Math.round(dmg)+"% HP).");
            if(target.hp===0){
              target.mesh.visible=false;
              logMsg("OVERRUN/SUNK: "+target.name+" eliminated by "+u.name+"!");
            }
          }else{
            launchMissile(u,target,true);
          }
        }
      }
    });
  }

  for(let i=explosions.length-1;i>=0;i--){
    const ex=explosions[i];
    ex.age += 0.04 * simSpd;
    ex.mesh.scale.setScalar(1+ex.age*4);
    ex.mesh.material.opacity=Math.max(0,1-ex.age);
    if(ex.age>=1){
      simGroup.remove(ex.mesh);
      explosions.splice(i,1);
    }
  }

  if(now-lastUI>400){
    updateHUD();
    if(state==="ENGAGED") renderRegistry();
    lastUI=now;
  }

  renderer.render(scene,camera);
}
